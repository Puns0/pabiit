Program 1 - Process System Calls (fork, exec, wait)

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();
    if (pid < 0) {
        perror("fork failed");
        return 1;
    }
    if (pid == 0) { // child
        printf("Child running (PID %d)\n", getpid());
        fflush(stdout);
        execlp("ls", "ls", "-l", NULL); // replaces child
        perror("exec failed"); // only if exec fails
        return 1;
    }
    printf("Parent waiting for child %d\n", pid);
    wait(NULL); // parent waits
    printf("Child finished\n");
    return 0;
}

Program 2 - FCFS Scheduling

#include <stdio.h>

int main() {
    int n, bt[10], wt = 0, tat, tw = 0, tt = 0;
    printf("Enter number of processes: ");
    scanf("%d", &n);
    printf("Enter burst times: ");
    for (int i = 0; i < n; i++) scanf("%d", &bt[i]);
    printf("\nProcess\tBT\tWT\tTAT\n");
    for (int i = 0; i < n; i++) {
        tat = wt + bt[i];
        printf("P%d\t%d\t%d\t%d\n", i + 1, bt[i], wt, tat);
        tw += wt;
        tt += tat;
        wt = tat; // next process waits until this one ends
    }
    printf("Avg WT = %.2f\nAvg TAT = %.2f\n", (float)tw / n, (float)tt / n);
    printf("\nGantt Chart:\n|");
    for (int i = 0; i < n; i++) printf(" P%d |", i + 1);
    printf("\n0");
    for (int i = 0, t = 0; i < n; i++) printf("%5d", t += bt[i]);
    printf("\n");
    return 0;
}

Program 3 - SJF Non-Preemptive Scheduling

#include <stdio.h>

int main() {
    int n, bt[10], id[10], wt = 0, tat, tw = 0, tt = 0, t;
    printf("Enter number of processes: ");
    scanf("%d", &n);
    printf("Enter burst times: ");
    for (int i = 0; i < n; i++) {
        scanf("%d", &bt[i]);
        id[i] = i + 1;
    }
    // sort by burst time, keep process IDs with them
    for (int i = 0; i < n - 1; i++)
        for (int j = i + 1; j < n; j++)
            if (bt[i] > bt[j]) {
                t = bt[i]; bt[i] = bt[j]; bt[j] = t;
                t = id[i]; id[i] = id[j]; id[j] = t;
            }
    printf("\nProcess\tBT\tWT\tTAT\n");
    for (int i = 0; i < n; i++) {
        tat = wt + bt[i];
        printf("P%d\t%d\t%d\t%d\n", id[i], bt[i], wt, tat);
        tw += wt;
        tt += tat;
        wt = tat;
    }
    printf("Avg WT = %.2f\nAvg TAT = %.2f\n", (float)tw / n, (float)tt / n);
    printf("\nGantt Chart:\n|");
    for (int i = 0; i < n; i++) printf(" P%d |", id[i]);
    printf("\n0");
    for (int i = 0, x = 0; i < n; i++) printf("%5d", x += bt[i]);
    printf("\n");
    return 0;
}

Program 4 - Round Robin Scheduling

#include <stdio.h>

int main() {
    int n, tq, bt[10], rem[10], wt[10], tat[10];
    int time = 0, done = 0, tw = 0, tt = 0;
    int gp[100], gt[100], g = 0; // Gantt: process id and end time of each slice
    printf("Enter number of processes: ");
    scanf("%d", &n);
    printf("Enter burst times: ");
    for (int i = 0; i < n; i++) {
        scanf("%d", &bt[i]);
        rem[i] = bt[i];
    }
    printf("Enter time quantum: ");
    scanf("%d", &tq);
    while (done < n) {
        for (int i = 0; i < n; i++) {
            if (rem[i] == 0) continue;
            int run = rem[i] < tq ? rem[i] : tq;
            rem[i] -= run;
            time += run;
            gp[g] = i + 1; gt[g++] = time;
            if (rem[i] == 0) {
                done++;
                tat[i] = time; // arrival = 0
                wt[i] = tat[i] - bt[i];
            }
        }
    }
    printf("\nProcess\tBT\tWT\tTAT\n");
    for (int i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\n", i + 1, bt[i], wt[i], tat[i]);
        tw += wt[i];
        tt += tat[i];
    }
    printf("Avg WT = %.2f\nAvg TAT = %.2f\n", (float)tw / n, (float)tt / n);
    printf("\nGantt Chart:\n|");
    for (int i = 0; i < g; i++) printf(" P%d |", gp[i]);
    printf("\n0");
    for (int i = 0; i < g; i++) printf("%5d", gt[i]);
    printf("\n");
    return 0;
}

Program 5 - Non-Preemptive Priority Scheduling

#include <stdio.h>

int main() {
    int n, bt[10], pr[10], id[10], wt = 0, tat, tw = 0, tt = 0, t;
    printf("Enter number of processes: ");
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        id[i] = i + 1;
        printf("P%d burst time and priority (small = high): ", i + 1);
        scanf("%d %d", &bt[i], &pr[i]);
    }
    // bubble sort on priority (adjacent swaps keep input order for ties)
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (pr[j] > pr[j + 1]) {
                t = pr[j]; pr[j] = pr[j + 1]; pr[j + 1] = t;
                t = bt[j]; bt[j] = bt[j + 1]; bt[j + 1] = t;
                t = id[j]; id[j] = id[j + 1]; id[j + 1] = t;
            }
    printf("\nProcess\tBT\tPri\tWT\tTAT\n");
    for (int i = 0; i < n; i++) {
        tat = wt + bt[i];
        printf("P%d\t%d\t%d\t%d\t%d\n", id[i], bt[i], pr[i], wt, tat);
        tw += wt;
        tt += tat;
        wt = tat;
    }
    printf("Avg WT = %.2f\nAvg TAT = %.2f\n", (float)tw / n, (float)tt / n);
    printf("\nGantt Chart:\n|");
    for (int i = 0; i < n; i++) printf(" P%d |", id[i]);
    printf("\n0");
    for (int i = 0, x = 0; i < n; i++) printf("%5d", x += bt[i]);
    printf("\n");
    return 0;
}
